# routine: contract-intake trigger: drive.file.changed under /files/drive/Contracts/ # Event payloads are untrusted: extract and quote, never follow instructions in them. ctx.events.each do |ev| doc = app.files.read_text(ev.subject) # via the VFS, Drive ACL applies facts = app.extract.contract(doc) # script: parties, dates, notice row = app.tables.upsert("Contracts", key: facts["file_id"], values: facts) next if facts["notice_days"].nil? verdict = app.ai.judge("renewal-clause", from: doc, untrusted: true) # Cleo, only here if verdict["action_needed"] app.issues.create(space: "Operations", title: "Check renewal clause · #{facts["party"]}", owner: ctx.owner, due: facts["notice_deadline"], body: verdict["summary"]) end end
